% Patch: make listings captions compatible with NKThesis \@makecaption[type]{num}{text}
% NKThesis uses TeX delimited syntax: \long\def\@makecaption[#1]#2#3
% listings calls \@makecaption{num}{text} without the [type] delimiter.
% \@ifnextchar[ detects which form is used and routes accordingly.
\makeatletter
\let\NKT@orig@makecaption\@makecaption
\def\@makecaption{%
  \@ifnextchar[%
    {\NKT@orig@makecaption}%
    {\NKT@lst@makecaption}%
}
\long\def\NKT@lst@makecaption#1#2{%
  % listings calls without [type]; use [figure] — the only safe default since
  % NKThesis only defines \belowcaptionskip@figure and \belowcaptionskip@table
  \NKT@orig@makecaption[figure]{#1}{#2}%
}
\makeatother

\makeatletter
% 1. Patch \@schapter so all \chapter*{} entries (List of Figures,
%    List of Tables, Abbreviations, References, Acknowledgements,
%    Resume) get a correct hyperref anchor before \addcontentsline.
\def\@schapter#1{%
  \if@twocolumn
    \@topnewpage[\@makeschapterhead{#1}]%
  \else
    \@makeschapterhead{#1}%
    \@afterheading
    \@nobreakfalse
  \fi
  \if@tocline
    \phantomsection
    \addcontentsline{toc}{chapter}{\protect\numberline{#1}{}}%
  \fi
  \markboth{#1}{#1}}
% 2. Patch \zhaiyao (Chinese abstract) — add \phantomsection
%    after \clearpage but before \addcontentsline.
\def\zhaiyao{%
  \clearpage
  \phantomsection
  \vspace*{24pt}
  \centerline{\zihaoxiaoer\rmfamily\bfseries\jiacu 摘要}
  \addcontentsline{toc}{chapter}{\protect\numberline{摘要}{}}
  \markboth{摘要}{摘要}
  \vspace*{18pt}
  \fontsize{12}{20}\selectfont}
\let\endzhaiyao\relax
% 3. Patch \abstract (English abstract) — same treatment.
\def\abstract{%
  \clearpage
  \phantomsection
  \vspace*{24pt}
  \centerline{\zihaoxiaoer\usefont{OT1}{phv}{bx}{n}Abstract}
  \addcontentsline{toc}{chapter}{\protect\numberline{Abstract}{}}
  \markboth{Abstract}{Abstract}
  \vspace*{18pt}
  \fontsize{12}{20}\selectfont}
\let\endabstract\relax
\makeatother
